def digest(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def obj(fields):return {'type':'object','properties':fields,'required':list(fields),'additionalProperties':False}
def arr(items):return {'type':'array','items':items}
def enum(values):return {'type':'string','enum':values}
text={'type':'string','minLength':1}
source_ids=[b['block_id'] for b in plan['original_context_blocks']]

context=json.dumps(plan['original_context_blocks'],ensure_ascii=False,separators=(',',':'))

def write_spec(phase,prompt,schema,budget,model='gpt-5.6-luna'):
    spec={'phase':phase,'model':model,'prompt':prompt,'json_schema':schema,'format_name':phase.replace('-','_'),
          'max_output_tokens':budget,'max_response_bytes':4*1024*1024,'wall_clock_limit_seconds':360,
          'source_sha256':request['source_sha256'],'normalized_sha256':request['normalized_sha256'],
          'input_files':[{'role':'source_plan','path':'plan.json','sha256':digest(ROOT/'plan.json')}],
          'source_ids':source_ids,'output_kind':'plain_text' if schema is None else 'json',
          'not_a_production_artifact':True}
    (ROOT/(phase+'-spec.json')).write_text(json.dumps(spec,ensure_ascii=False,indent=2)+'\n')
    return spec

def decode(raw):
    return raw if spec['output_kind']=='plain_text' else json.loads(raw)

def validate(value):
    if spec['output_kind']=='plain_text':
        if not isinstance(value,str) or not value.strip():return ['empty plain prose']
        if value.lstrip().startswith(('```','{','[')):return ['expected plain prose, received wrapper']
        return []
    errors=[f'{list(e.path)}: {e.message}' for e in Draft202012Validator(spec['json_schema']).iter_errors(value)]
    if errors:return errors
    if spec['phase']=='source-facts':
        entities={e['ref']:e for e in value['entities']}
        if len(entities)!=len(value['entities']):errors.append('duplicate entity refs')
        fact_ids=[f['ref'] for f in value['facts']]
        if len(fact_ids)!=len(set(fact_ids)):errors.append('duplicate fact refs')
        for entity in value['entities']:
            if not entity['source_ids']:errors.append(entity['ref']+': missing source ids')
        for fact in value['facts']:
            subject=entities.get(fact['subject_ref'])
            if subject is None or subject['name']!=fact['subject_name']:errors.append(fact['ref']+': subject ref/name mismatch')
            if set(fact['other_refs'])-set(entities):errors.append(fact['ref']+': unknown participant ref')
            if not fact['source_ids']:errors.append(fact['ref']+': missing source ids')

            if t['basis']=='unknown' and t['label'] is not None:errors.append(fact['ref']+': unknown time must have null label')
            if t['basis']!='unknown' and (not t['label'] or not t['source_ids']):errors.append(fact['ref']+': known time missing source ids/label')
